# ============================================================
# SOFTGEST WEB - PROTEÇÃO DE ARQUIVOS
# ============================================================

# Ativar motor de reescrita
RewriteEngine On

# ============================================================
# PROTEÇÃO CONTRA LISTAGEM DE DIRETÓRIOS
# ============================================================
Options -Indexes

# ============================================================
# PÁGINA INICIAL
# ============================================================
DirectoryIndex index.php login.php

# ============================================================
# PROTEÇÃO DE ARQUIVOS SENSÍVEIS
# ============================================================
<FilesMatch "\.(sql|log|ini|config|json|yml|yaml|md|sh|bat|py)$">
    Order allow,deny
    Deny from all
</FilesMatch>

# ============================================================
# PROTEÇÃO DE ARQUIVOS DE CONFIGURAÇÃO
# ============================================================
<Files "database.php">
    Order allow,deny
    Deny from all
</Files>

<Files "security.php">
    Order allow,deny
    Deny from all
</Files>

# ============================================================
# PROTEÇÃO CONTRA ACESSO DIRETO A PASTAS SENSÍVEIS
# ============================================================
# APENAS PASTAS QUE NÃO DEVEM SER ACESSADAS DIRETAMENTE
RewriteRule ^config/.*$ - [F,L]
RewriteRule ^sql/.*$ - [F,L]
RewriteRule ^protect/.*$ - [F,L]

# ============================================================
# PERMITIR ACESSO A PASTAS DE PERFIS (Dashboards)
# ============================================================
# PERMITIR acesso a todas as pastas de perfil
RewriteRule ^(admin|diretor|vice_diretor|coordenador|coordenador_pedagogico|supervisor|orientador|pedagogo|psicopedagogo|professor|docente|instrutor|monitor|tutor|auxiliar_docente|aluno|estudante|bolsista|estagiario|secretaria|financeiro|contador|rh|atendimento|recepcionista|vigilante|motorista|aux_limpeza|jardineiro|manutencao|porteiro|merendeira|aux_servicos|encarregado|responsavel|pai|mae|tutor_legal|gerente|gestor)/dashboard\.php$ - [L]

# PERMITIR acesso a includes (header/footer)
RewriteRule ^includes/.*\.(php|css|js)$ - [L]

# NEGAR acesso a outros arquivos dentro de includes
RewriteRule ^includes/.*\.(sql|log|ini|config)$ - [F,L]

# ============================================================
# PERMITIR SSE (SERVER-SENT EVENTS)
# ============================================================
<Files "chamada_stream.php">
    # Desabilitar cache para SSE
    <IfModule mod_headers.c>
        Header set Cache-Control "no-cache, no-store, must-revalidate"
        Header set Pragma "no-cache"
        Header set Expires "0"
        Header set X-Accel-Buffering "no"
        Header set Content-Type "text/event-stream"
    </IfModule>
</Files>

# ============================================================
# PROTEÇÃO CONTRA HOTLINKING (IMAGENS)
# ============================================================
RewriteCond %{HTTP_REFERER} !^$
RewriteCond %{HTTP_REFERER} !^http(s)?://(www\.)?localhost [NC]
RewriteRule \.(jpg|jpeg|png|gif|svg|webp|ico)$ - [F,L]

# ============================================================
# PREVENÇÃO DE INJEÇÃO DE CÓDIGO
# ============================================================
RewriteCond %{QUERY_STRING} (\<|%3C).*script.*(\>|%3E) [NC,OR]
RewriteCond %{QUERY_STRING} GLOBALS(=|\[|\%[0-9A-Z]{0,2}) [OR]
RewriteCond %{QUERY_STRING} _REQUEST(=|\[|\%[0-9A-Z]{0,2})
RewriteRule .* index.php [F,L]

# ============================================================
# BLOQUEAR ACESSO A ARQUIVOS OCULTOS
# ============================================================
RedirectMatch 403 /\..*$

# ============================================================
# COMPRESSÃO GZIP
# ============================================================
<IfModule mod_deflate.c>
    # Não comprimir SSE
    SetEnvIfNoCase Request_URI "chamada_stream\.php$" no-gzip dont-vary
    AddOutputFilterByType DEFLATE text/html text/css text/javascript application/javascript application/json text/plain text/xml
</IfModule>

# ============================================================
# CACHE DE ARQUIVOS ESTÁTICOS
# ============================================================
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/jpg "access plus 1 year"
    ExpiresByType image/jpeg "access plus 1 year"
    ExpiresByType image/gif "access plus 1 year"
    ExpiresByType image/png "access plus 1 year"
    ExpiresByType text/css "access plus 1 month"
    ExpiresByType application/javascript "access plus 1 month"
    # SSE não deve ser cacheados
    ExpiresByType text/event-stream "access plus 0 seconds"
</IfModule>

# ============================================================
# SEGURANÇA DE CABEÇALHOS
# ============================================================
<IfModule mod_headers.c>
    Header set X-Content-Type-Options "nosniff"
    Header set X-Frame-Options "DENY"
    Header set X-XSS-Protection "1; mode=block"
    Header set Referrer-Policy "no-referrer-when-downgrade"
</IfModule>

# ============================================================
# TIMEOUT PARA SSE (CONEXÃO LONGA)
# ============================================================
<IfModule mod_fcgid.c>
    FcgidIOTimeout 600
    FcgidBusyTimeout 600
</IfModule>

<IfModule mod_php.c>
    php_value max_execution_time 600
    php_value max_input_time 600
    php_value memory_limit 256M
</IfModule>

# ============================================================
# REDIRECIONAR 404 PARA PÁGINA PERSONALIZADA
# ============================================================
ErrorDocument 404 /softgest_web/404.php
ErrorDocument 403 /softgest_web/403.php
