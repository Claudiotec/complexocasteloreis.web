$itens = [];
$totais = [
    'receita' => 0,
    'despesa' => 0,
    'liquido' => 0
];

try {
    // Busca o orçamento
    $stmt = $pdo->prepare("SELECT * FROM orcamento WHERE ano = ?");
    $stmt->execute([$ano]);
    $orcamento = $stmt->fetch();
    
    if ($orcamento) {
        // Busca os itens do orçamento
        $stmt = $pdo->prepare("
            SELECT oi.*, pc.codigo, pc.nome as conta_nome, pc.tipo
            FROM orcamento_itens oi
            JOIN plano_contas pc ON oi.conta_id = pc.id
            WHERE oi.orcamento_id = ?
            ORDER BY pc.codigo
        ");
        $stmt->execute([$orcamento['id']]);
        $itens = $stmt->fetchAll();
        
        $totais['receita'] = $orcamento['total_receita'] ?? 0;
        $totais['despesa'] = $orcamento['total_despesa'] ?? 0;
        $totais['liquido'] = $orcamento['total_liquido'] ?? 0;
    }
} catch (Exception $e) {
