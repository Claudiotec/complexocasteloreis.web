-- =============================================
-- 2. RECRIAR TABELA FORCA_TRABALHO
-- =============================================

CREATE TABLE forca_trabalho (
    id INT PRIMARY KEY AUTO_INCREMENT,
    numero_ordem VARCHAR(10),
    numero_instituicao VARCHAR(20),
    numero_agente VARCHAR(20),
    nome_completo VARCHAR(200) NOT NULL,
    categoria_actual VARCHAR(100),
    instituicao VARCHAR(200),
    funcao VARCHAR(100),
    disciplina VARCHAR(100),
    formacao_disciplina VARCHAR(50),
    data_inicio_funcao DATE,
    data_inicio_instituicao DATE,
    bi VARCHAR(30) UNIQUE,
    data_nascimento DATE,
    genero CHAR(1),
    habilitacoes VARCHAR(100),
    especialidade_medio VARCHAR(100),
    especialidade_superior VARCHAR(100),
    contacto VARCHAR(20),
    municipio VARCHAR(100),
    email VARCHAR(100),
    status ENUM('ativo', 'inativo', 'ferias') DEFAULT 'ativo',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_nome (nome_completo),
    INDEX idx_status (status),
    INDEX idx_bi (bi),
    INDEX idx_agente (numero_agente)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- 3. RECRIAR TABELA HORARIOS_TRABALHO
-- =============================================

CREATE TABLE horarios_trabalho (
    id INT PRIMARY KEY AUTO_INCREMENT,
    funcionario_id INT NOT NULL,
    dia_semana ENUM('segunda', 'terca', 'quarta', 'quinta', 'sexta', 'sabado', 'domingo') NOT NULL,
    hora_entrada TIME NOT NULL,
    hora_saida TIME NOT NULL,
    hora_intervalo_inicio TIME NOT NULL,
    hora_intervalo_fim TIME NOT NULL,
    carga_horaria DECIMAL(5,2),
    turno ENUM('manha', 'tarde', 'noite', 'integral') DEFAULT 'manha',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (funcionario_id) REFERENCES forca_trabalho(id) ON DELETE CASCADE,
    INDEX idx_funcionario (funcionario_id),
    INDEX idx_dia (dia_semana),
    UNIQUE KEY unique_horario (funcionario_id, dia_semana)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- 4. RECRIAR TABELA QR_CODES_DIARIOS
-- =============================================

CREATE TABLE qr_codes_diarios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    funcionario_id INT NOT NULL,
    data DATE NOT NULL,
    qr_code TEXT,
    token VARCHAR(64) UNIQUE,
    expiracao TIMESTAMP,
    gerado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (funcionario_id) REFERENCES forca_trabalho(id) ON DELETE CASCADE,
    INDEX idx_funcionario_data (funcionario_id, data),
    INDEX idx_token (token),
    INDEX idx_expiracao (expiracao)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =============================================
-- 5. RECRIAR TABELA PRESENCA_QR
-- =============================================

CREATE TABLE presenca_qr (
    id INT PRIMARY KEY AUTO_INCREMENT,
    funcionario_id INT NOT NULL,
    data DATE NOT NULL,
    hora_entrada TIME,
    hora_saida TIME,
    status ENUM('presente', 'ausente', 'atraso', 'justificado') DEFAULT 'ausente',
    qr_code TEXT,
    qr_scanned BOOLEAN DEFAULT FALSE,
    qr_scan_time TIMESTAMP NULL,
    horas_trabalhadas DECIMAL(5,2),
    observacao TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (funcionario_id) REFERENCES forca_trabalho(id) ON DELETE CASCADE,
    INDEX idx_funcionario_data (funcionario_id, data),
    INDEX idx_status (status),
    INDEX idx_data (data),
    UNIQUE KEY unique_presenca_dia (funcionario_id, data)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- TABELA DE MENSAGENS
-- ============================================
CREATE TABLE IF NOT EXISTS mensagens (
    id INT PRIMARY KEY AUTO_INCREMENT,
    remetente_id INT NOT NULL,
    destinatario_id INT NULL,
    assunto VARCHAR(200) NOT NULL,
    mensagem TEXT NOT NULL,
    tipo ENUM('publica', 'privada') DEFAULT 'publica',
    status ENUM('lida', 'nao_lida') DEFAULT 'nao_lida',
    data_envio DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_leitura DATETIME NULL,
    respondendo_a INT NULL,
    FOREIGN KEY (remetente_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (destinatario_id) REFERENCES usuarios(id) ON DELETE CASCADE,
    FOREIGN KEY (respondendo_a) REFERENCES mensagens(id) ON DELETE SET NULL,
    INDEX idx_destinatario (destinatario_id),
    INDEX idx_data_envio (data_envio)
);

-- ============================================
-- TABELA DE ARQUIVOS ANEXADOS
-- ============================================
CREATE TABLE IF NOT EXISTS mensagens_anexos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    mensagem_id INT NOT NULL,
    nome_original VARCHAR(255) NOT NULL,
    nome_arquivo VARCHAR(255) NOT NULL,
    caminho VARCHAR(500) NOT NULL,
    tipo_arquivo VARCHAR(100) NOT NULL,
    tamanho INT NOT NULL,
    data_upload DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (mensagem_id) REFERENCES mensagens(id) ON DELETE CASCADE,
    INDEX idx_mensagem (mensagem_id)
);




-- ============================================
-- CRIAR TABELA horarios_trabalho
-- ============================================
CREATE TABLE IF NOT EXISTS horarios_trabalho (
    id INT AUTO_INCREMENT PRIMARY KEY,
    funcionario_id INT NOT NULL,
    dia_semana VARCHAR(20) NOT NULL,
    hora_entrada TIME NOT NULL,
    hora_saida TIME NOT NULL,
    hora_intervalo_inicio TIME NOT NULL,
    hora_intervalo_fim TIME NOT NULL,
    carga_horaria DECIMAL(5,2) NOT NULL,
    turno VARCHAR(20) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_funcionario_id (funcionario_id),
    INDEX idx_dia_semana (dia_semana),
    INDEX idx_turno (turno),
    
    FOREIGN KEY (funcionario_id) REFERENCES forca_trabalho(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
