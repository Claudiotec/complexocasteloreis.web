Script para REMOVER as 3 constraints:
sql
-- REMOVER constraints das 3 tabelas
ALTER TABLE `faltas` DROP FOREIGN KEY `faltas_ibfk_1`;
ALTER TABLE `folha_pagamento` DROP FOREIGN KEY `folha_pagamento_ibfk_1`;
ALTER TABLE `presencas` DROP FOREIGN KEY `presencas_ibfk_1`;
Depois de remover, atualize o funcionário:
sql
-- Agora o UPDATE vai funcionar!
UPDATE `funcionarios` SET `id` = 5 WHERE `funcionarios`.`id` = 7;
Verifique se funcionou:
sql
-- Verificar se o ID foi alterado
SELECT * FROM `funcionarios` WHERE `id` = 5;
(OPCIONAL) Recriar as constraints depois:
sql
-- RECRIAR constraints
ALTER TABLE `faltas` 
ADD CONSTRAINT `faltas_ibfk_1` 
FOREIGN KEY (`funcionario_id`) 
REFERENCES `funcionarios` (`id`) 
ON DELETE CASCADE;
